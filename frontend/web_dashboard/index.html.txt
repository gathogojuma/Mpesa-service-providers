    <script>
        const API_URL = 'http://localhost:8000';
        let token = localStorage.getItem('token');

        if (token) showDashboard();

        async function login() {
            const phone = document.getElementById('phone').value;
            const pin = document.getElementById('pin').value;
            try {
                const res = await fetch(`${API_URL}/api/auth/login`, {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify({phone, pin})
                });
                const data = await res.json();
                if (res.ok) {
                    token = data.access_token;
                    localStorage.setItem('token', token);
                    showDashboard();
                } else {
                    document.getElementById('loginError').textContent = data.detail || 'Login failed';
                }
            } catch (e) {
                document.getElementById('loginError').textContent = 'Server error';
            }
        }

        function logout() {
            localStorage.removeItem('token');
            token = null;
            location.reload();
        }

        function showDashboard() {
            document.getElementById('loginPage').classList.add('hidden');
            document.getElementById('dashboardPage').classList.remove('hidden');
            loadDashboard();
            setInterval(loadDashboard, 10000);
        }

        async function loadDashboard() {
            const res = await fetch(`${API_URL}/api/manager/dashboard`, {
                headers: {'Authorization': `Bearer ${token}`}
            });
            if (!res.ok) { logout(); return; }
            const data = await res.json();

            document.getElementById('totalRevenue').textContent = `KES ${data.total_revenue_today.toFixed(2)}`;
            document.getElementById('totalTransactions').textContent = data.total_transactions_today;
            document.getElementById('pendingCount').textContent = data.pending_count;
            document.getElementById('flaggedCount').textContent = data.flagged_count;

            document.getElementById('serverStats').innerHTML = data.server_stats.map(s => `
                <tr><td>${s.name}</td><td>KES ${s.total_sales.toFixed(2)}</td><td>${s.transaction_count}</td></tr>
            `).join('');

            document.getElementById('flaggedTable').innerHTML = data.flagged_transactions.map(t => `
                <tr>
                    <td>${t.staff_name}</td>
                    <td>KES ${t.amount.toFixed(2)}</td>
                    <td>${t.failure_reason || 'Unknown'}</td>
                    <td><button class="btn btn-warning" onclick="reconcile('${t.id}')">Fix</button></td>
                </tr>
            `).join('');
        }

        async function reconcile(id) {
            await fetch(`${API_URL}/api/manager/reconcile`, {
                method: 'POST',
                headers: {'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json'},
                body: JSON.stringify({transaction_id: id})
            });
            loadDashboard();
        }
    </script>
